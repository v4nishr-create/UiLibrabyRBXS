Librairie Luau — animations d'UI

Des interfaces Roblox qui bougent, sans réinventer le tween.

UIAnim est un module Luau unique qui ajoute shake, pulse, bounce, fade, slide, flash, rainbow et typewriter à n'importe quel GuiObject. Un require, un appel de fonction, c'est animé.

Mise en route

Installation en trois étapes

Aucun gestionnaire de paquets : UIAnim est un seul ModuleScript à copier dans votre jeu.

1

Créer le ModuleScript

Dans l'Explorer, ajoutez un ModuleScript nommé exactement UIAnim dans ReplicatedStorage (accessible client et serveur, mais les animations elles-mêmes doivent être lancées depuis un LocalScript).

2

Coller le script

Copiez le contenu de la section Script complet ci-dessous dans ce ModuleScript, en remplaçant le contenu par défaut.

3

Importer et utiliser

Depuis n'importe quel LocalScript, require le module et appelez une fonction sur votre GuiObject.

lua
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local UIAnim = require(ReplicatedStorage:WaitForChild("UIAnim"))

local errorFrame = script.Parent:WaitForChild("ErrorPanel")
UIAnim.Shake(errorFrame, { intensity = 10, duration = 0.5 })

Code source

Le script complet

Un seul fichier, aucune dépendance externe. Copiez-le tel quel dans votre ModuleScript UIAnim.

lua — UIAnim.lua 
--!strict
--[[
	UIAnim — petite librairie d'animations d'UI pour Roblox Studio.

	Installation :
		1. Créez un ModuleScript nommé "UIAnim" (par ex. dans ReplicatedStorage).
		2. Collez tout ce fichier dedans.
		3. Dans un LocalScript : local UIAnim = require(game.ReplicatedStorage.UIAnim)

	Toutes les fonctions prennent un GuiObject (Frame, ImageLabel, TextLabel, TextButton...)
	et une table d'options optionnelle. Elles annulent automatiquement toute animation
	en cours sur le même objet pour éviter les conflits.
]]

local TweenService = game:GetService("TweenService")
local RunService = game:GetService("RunService")

local UIAnim = {}

-- Un seul effet actif par objet à la fois : on stocke une fonction d'annulation.
local activeEffects: { [Instance]: () -> () } = {}

local function setActive(obj: Instance, cancelFn: () -> ())
	local existing = activeEffects[obj]
	if existing then
		existing()
	end
	activeEffects[obj] = cancelFn
end

local function clearIfCurrent(obj: Instance, cancelFn: () -> ())
	if activeEffects[obj] == cancelFn then
		activeEffects[obj] = nil
	end
end

--- Arrête l'animation en cours sur un objet (restaure un état cohérent).
function UIAnim.Stop(obj: Instance)
	local existing = activeEffects[obj]
	if existing then
		existing()
		activeEffects[obj] = nil
	end
end

--------------------------------------------------------------------------
-- Shake : secoue l'objet (ex: mauvaise saisie, erreur, impact)
--------------------------------------------------------------------------
export type ShakeOptions = {
	intensity: number?, -- amplitude en pixels, défaut 6
	duration: number?, -- durée totale en secondes, défaut 0.4
}

function UIAnim.Shake(obj: GuiObject, options: ShakeOptions?)
	options = options or {}
	local intensity = options.intensity or 6
	local duration = options.duration or 0.4

	local originalPosition = obj.Position
	local elapsed = 0
	local conn: RBXScriptConnection

	local function cancel()
		if conn then
			conn:Disconnect()
		end
		obj.Position = originalPosition
	end

	conn = RunService.RenderStepped:Connect(function(dt)
		elapsed += dt
		if elapsed >= duration then
			cancel()
			clearIfCurrent(obj, cancel)
			return
		end
		local falloff = 1 - (elapsed / duration)
		local offsetX = (math.random() * 2 - 1) * intensity * falloff
		local offsetY = (math.random() * 2 - 1) * intensity * falloff
		obj.Position = originalPosition + UDim2.fromOffset(offsetX, offsetY)
	end)

	setActive(obj, cancel)
end

--------------------------------------------------------------------------
-- Wiggle : secoue en rotation (ex: icône qui attire l'attention)
--------------------------------------------------------------------------
export type WiggleOptions = {
	angle: number?, -- degrés max, défaut 8
	duration: number?, -- défaut 0.5
}

function UIAnim.Wiggle(obj: GuiObject, options: WiggleOptions?)
	options = options or {}
	local angle = options.angle or 8
	local duration = options.duration or 0.5

	local originalRotation = obj.Rotation
	local elapsed = 0
	local conn: RBXScriptConnection

	local function cancel()
		if conn then
			conn:Disconnect()
		end
		obj.Rotation = originalRotation
	end

	conn = RunService.RenderStepped:Connect(function(dt)
		elapsed += dt
		if elapsed >= duration then
			cancel()
			clearIfCurrent(obj, cancel)
			return
		end
		local falloff = 1 - (elapsed / duration)
		obj.Rotation = originalRotation + math.sin(elapsed * 40) * angle * falloff
	end)

	setActive(obj, cancel)
end

--------------------------------------------------------------------------
-- Pulse : grossit puis revient (ex: confirmation, notification)
--------------------------------------------------------------------------
export type PulseOptions = {
	scale: number?, -- facteur max, défaut 1.15
	duration: number?, -- durée d'un aller-retour, défaut 0.25
}

function UIAnim.Pulse(obj: GuiObject, options: PulseOptions?)
	options = options or {}
	local scaleFactor = options.scale or 1.15
	local duration = options.duration or 0.25

	local originalSize = obj.Size
	local grownSize = UDim2.new(
		originalSize.X.Scale * scaleFactor,
		originalSize.X.Offset * scaleFactor,
		originalSize.Y.Scale * scaleFactor,
		originalSize.Y.Offset * scaleFactor
	)

	local up = TweenService:Create(obj, TweenInfo.new(duration / 2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Size = grownSize })
	local down = TweenService:Create(obj, TweenInfo.new(duration / 2, Enum.EasingStyle.Quad, Enum.EasingDirection.In), { Size = originalSize })

	local function cancel()
		up:Cancel()
		down:Cancel()
		obj.Size = originalSize
	end

	up.Completed:Connect(function(state)
		if state == Enum.PlaybackState.Completed then
			down:Play()
		end
	end)
	down.Completed:Connect(function()
		clearIfCurrent(obj, cancel)
	end)

	setActive(obj, cancel)
	up:Play()
end

--------------------------------------------------------------------------
-- FadeIn / FadeOut : apparition / disparition en douceur
--------------------------------------------------------------------------
export type FadeOptions = {
	duration: number?, -- défaut 0.3
}

local function getTransparencyProperty(obj: Instance): string?
	if obj:IsA("Frame") or obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
		return "BackgroundTransparency"
	elseif obj:IsA("TextLabel") or obj:IsA("TextButton") then
		return "TextTransparency"
	end
	return nil
end

function UIAnim.FadeIn(obj: GuiObject, options: FadeOptions?)
	options = options or {}
	local duration = options.duration or 0.3

	obj.Visible = true
	local prop = getTransparencyProperty(obj)
	local goal: { [string]: any } = {}
	if prop then
		goal[prop] = 0
	end

	local tween = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), goal)
	local function cancel()
		tween:Cancel()
	end
	tween.Completed:Connect(function()
		clearIfCurrent(obj, cancel)
	end)
	setActive(obj, cancel)
	tween:Play()
end

function UIAnim.FadeOut(obj: GuiObject, options: FadeOptions?)
	options = options or {}
	local duration = options.duration or 0.3

	local prop = getTransparencyProperty(obj)
	local goal: { [string]: any } = {}
	if prop then
		goal[prop] = 1
	end

	local tween = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Sine, Enum.EasingDirection.In), goal)
	local function cancel()
		tween:Cancel()
	end
	tween.Completed:Connect(function(state)
		if state == Enum.PlaybackState.Completed then
			obj.Visible = false
		end
		clearIfCurrent(obj, cancel)
	end)
	setActive(obj, cancel)
	tween:Play()
end

--------------------------------------------------------------------------
-- Bounce : rebond élastique à l'apparition
--------------------------------------------------------------------------
export type BounceOptions = {
	duration: number?, -- défaut 0.6
}

function UIAnim.Bounce(obj: GuiObject, options: BounceOptions?)
	options = options or {}
	local duration = options.duration or 0.6

	local targetSize = obj.Size
	obj.Size = UDim2.new(targetSize.X.Scale * 0.5, targetSize.X.Offset * 0.5, targetSize.Y.Scale * 0.5, targetSize.Y.Offset * 0.5)

	local tween = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out), { Size = targetSize })
	local function cancel()
		tween:Cancel()
		obj.Size = targetSize
	end
	tween.Completed:Connect(function()
		clearIfCurrent(obj, cancel)
	end)
	setActive(obj, cancel)
	tween:Play()
end

--------------------------------------------------------------------------
-- SlideIn : entre depuis un bord de l'écran
--------------------------------------------------------------------------
export type SlideDirection = "Top" | "Bottom" | "Left" | "Right"
export type SlideOptions = {
	direction: SlideDirection?, -- défaut "Bottom"
	distance: number?, -- pixels, défaut 60
	duration: number?, -- défaut 0.35
}

function UIAnim.SlideIn(obj: GuiObject, options: SlideOptions?)
	options = options or {}
	local direction = options.direction or "Bottom"
	local distance = options.distance or 60
	local duration = options.duration or 0.35

	local targetPosition = obj.Position
	local offsetX, offsetY = 0, 0
	if direction == "Top" then
		offsetY = -distance
	elseif direction == "Bottom" then
		offsetY = distance
	elseif direction == "Left" then
		offsetX = -distance
	elseif direction == "Right" then
		offsetX = distance
	end

	obj.Position = targetPosition + UDim2.fromOffset(offsetX, offsetY)
	obj.Visible = true

	local tween = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Position = targetPosition })
	local function cancel()
		tween:Cancel()
		obj.Position = targetPosition
	end
	tween.Completed:Connect(function()
		clearIfCurrent(obj, cancel)
	end)
	setActive(obj, cancel)
	tween:Play()
end

--------------------------------------------------------------------------
-- Flash : clignote (ex: alerte, surbrillance temporaire)
--------------------------------------------------------------------------
export type FlashOptions = {
	color: Color3?, -- couleur du flash, défaut rouge
	count: number?, -- nombre de clignotements, défaut 2
	duration: number?, -- durée d'un clignotement, défaut 0.15
}

function UIAnim.Flash(obj: GuiObject, options: FlashOptions?)
	options = options or {}
	local flashColor = options.color or Color3.fromRGB(255, 70, 70)
	local count = options.count or 2
	local duration = options.duration or 0.15

	if not (obj:IsA("Frame") or obj:IsA("ImageLabel") or obj:IsA("ImageButton") or obj:IsA("TextButton") or obj:IsA("TextLabel")) then
		return
	end

	local originalColor = (obj :: any).BackgroundColor3
	local cancelled = false

	local function cancel()
		cancelled = true
		;(obj :: any).BackgroundColor3 = originalColor
	end
	setActive(obj, cancel)

	task.spawn(function()
		for _ = 1, count do
			if cancelled then
				return
			end
			local toFlash = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Sine), { BackgroundColor3 = flashColor })
			toFlash:Play()
			toFlash.Completed:Wait()
			if cancelled then
				return
			end
			local toOriginal = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Sine), { BackgroundColor3 = originalColor })
			toOriginal:Play()
			toOriginal.Completed:Wait()
		end
		clearIfCurrent(obj, cancel)
	end)
end

--------------------------------------------------------------------------
-- Typewriter : révèle un texte lettre par lettre
--------------------------------------------------------------------------
export type TypewriterOptions = {
	charsPerSecond: number?, -- défaut 30
}

function UIAnim.Typewriter(label: TextLabel | TextButton, fullText: string, options: TypewriterOptions?)
	options = options or {}
	local cps = options.charsPerSecond or 30

	label.Text = ""
	local cancelled = false
	local function cancel()
		cancelled = true
		label.Text = fullText
	end
	setActive(label, cancel)

	task.spawn(function()
		local charDelay = 1 / cps
		for i = 1, #fullText do
			if cancelled then
				return
			end
			label.Text = string.sub(fullText, 1, i)
			task.wait(charDelay)
		end
		clearIfCurrent(label, cancel)
	end)
end

--------------------------------------------------------------------------
-- Rainbow : fait défiler les couleurs (ex: effet néon/fun)
--------------------------------------------------------------------------
export type RainbowOptions = {
	speed: number?, -- cycles par seconde, défaut 0.3
	target: ("Background" | "Text" | "Stroke")?, -- défaut "Background"
}

function UIAnim.Rainbow(obj: GuiObject, options: RainbowOptions?)
	options = options or {}
	local speed = options.speed or 0.3
	local target = options.target or "Background"

	local conn: RBXScriptConnection
	local hue = 0

	local function cancel()
		if conn then
			conn:Disconnect()
		end
	end

	conn = RunService.RenderStepped:Connect(function(dt)
		hue = (hue + dt * speed) % 1
		local color = Color3.fromHSV(hue, 0.85, 1)
		if target == "Background" and (obj :: any):FindFirstAncestorOrSelf then
			pcall(function()
				(obj :: any).BackgroundColor3 = color
			end)
		elseif target == "Text" then
			pcall(function()
				(obj :: any).TextColor3 = color
			end)
		elseif target == "Stroke" then
			local stroke = obj:FindFirstChildOfClass("UIStroke")
			if stroke then
				stroke.Color = color
			end
		end
	end)

	setActive(obj, cancel)
end

return UIAnim
