Librairie Luau — animations d'UI

Des interfaces Roblox qui bougent, sans réinventer le tween.

UIAnim est un module Luau unique qui ajoute 23 animations — shake, pulse, bounce, glow, scramble, wave, glitch, et l'ouverture/fermeture de menus — à n'importe quel GuiObject ou TextLabel. Un require, un appel de fonction, c'est animé.

Mise en route

Installation en trois étapes

Aucun gestionnaire de paquets : UIAnim est un seul ModuleScript à copier dans votre jeu.

1

Créer le ModuleScript

Dans l'Explorer, ajoutez un ModuleScript nommé exactement UIAnim dans ReplicatedStorage (accessible client et serveur, mais les animations elles-mêmes doivent être lancées depuis un LocalScript).

2

Coller le script

Copiez le contenu de la section Script complet ci-dessous dans ce ModuleScript, en remplaçant le contenu par défaut.

3

Importer et utiliser

Depuis n'importe quel LocalScript, require le module et appelez une fonction sur votre GuiObject.

lua
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local UIAnim = require(ReplicatedStorage:WaitForChild("UIAnim"))

local errorFrame = script.Parent:WaitForChild("ErrorPanel")
UIAnim.Shake(errorFrame, { intensity = 10, duration = 0.5 })

Code source

Le script complet

Un seul fichier, aucune dépendance externe. Copiez-le tel quel dans votre ModuleScript UIAnim.

lua — UIAnim.lua 
--!strict
--[[
	UIAnim — petite librairie d'animations d'UI pour Roblox Studio.

	Installation :
		1. Créez un ModuleScript nommé "UIAnim" (par ex. dans ReplicatedStorage).
		2. Collez tout ce fichier dedans.
		3. Dans un LocalScript : local UIAnim = require(game.ReplicatedStorage.UIAnim)

	Toutes les fonctions prennent un GuiObject (Frame, ImageLabel, TextLabel, TextButton...)
	et une table d'options optionnelle. Elles annulent automatiquement toute animation
	en cours sur le même objet pour éviter les conflits.
]]

local TweenService = game:GetService("TweenService")
local RunService = game:GetService("RunService")

local UIAnim = {}

-- Un seul effet actif par objet à la fois : on stocke une fonction d'annulation.
local activeEffects: { [Instance]: () -> () } = {}

local function setActive(obj: Instance, cancelFn: () -> ())
	local existing = activeEffects[obj]
	if existing then
		existing()
	end
	activeEffects[obj] = cancelFn
end

local function clearIfCurrent(obj: Instance, cancelFn: () -> ())
	if activeEffects[obj] == cancelFn then
		activeEffects[obj] = nil
	end
end

--- Arrête l'animation en cours sur un objet (restaure un état cohérent).
function UIAnim.Stop(obj: Instance)
	local existing = activeEffects[obj]
	if existing then
		existing()
		activeEffects[obj] = nil
	end
end

--------------------------------------------------------------------------
-- Shake : secoue l'objet (ex: mauvaise saisie, erreur, impact)
--------------------------------------------------------------------------
export type ShakeOptions = {
	intensity: number?, -- amplitude en pixels, défaut 6
	duration: number?, -- durée totale en secondes, défaut 0.4
}

function UIAnim.Shake(obj: GuiObject, options: ShakeOptions?)
	options = options or {}
	local intensity = options.intensity or 6
	local duration = options.duration or 0.4

	local originalPosition = obj.Position
	local elapsed = 0
	local conn: RBXScriptConnection

	local function cancel()
		if conn then
			conn:Disconnect()
		end
		obj.Position = originalPosition
	end

	conn = RunService.RenderStepped:Connect(function(dt)
		elapsed += dt
		if elapsed >= duration then
			cancel()
			clearIfCurrent(obj, cancel)
			return
		end
		local falloff = 1 - (elapsed / duration)
		local offsetX = (math.random() * 2 - 1) * intensity * falloff
		local offsetY = (math.random() * 2 - 1) * intensity * falloff
		obj.Position = originalPosition + UDim2.fromOffset(offsetX, offsetY)
	end)

	setActive(obj, cancel)
end

-- Wiggle : secoue en rotation (ex: icône qui attire l'attention)
--------------------------------------------------------------------------
export type WiggleOptions = {
	angle: number?, -- degrés max, défaut 8
	duration: number?, -- défaut 0.5
}

function UIAnim.Wiggle(obj: GuiObject, options: WiggleOptions?)
	options = options or {}
	local angle = options.angle or 8
	local duration = options.duration or 0.5

	local originalRotation = obj.Rotation
	local elapsed = 0
	local conn: RBXScriptConnection

	local function cancel()
		if conn then
			conn:Disconnect()
		end
		obj.Rotation = originalRotation
	end

	conn = RunService.RenderStepped:Connect(function(dt)
		elapsed += dt
		if elapsed >= duration then
			cancel()
			clearIfCurrent(obj, cancel)
			return
		end
		local falloff = 1 - (elapsed / duration)
		obj.Rotation = originalRotation + math.sin(elapsed * 40) * angle * falloff
	end)

	setActive(obj, cancel)
end

--------------------------------------------------------------------------
-- Pulse : grossit puis revient (ex: confirmation, notification)
--------------------------------------------------------------------------
export type PulseOptions = {
	scale: number?, -- facteur max, défaut 1.15
	duration: number?, -- durée d'un aller-retour, défaut 0.25
}

function UIAnim.Pulse(obj: GuiObject, options: PulseOptions?)
	options = options or {}
	local scaleFactor = options.scale or 1.15
	local duration = options.duration or 0.25

	local originalSize = obj.Size
	local grownSize = UDim2.new(
		originalSize.X.Scale * scaleFactor,
		originalSize.X.Offset * scaleFactor,
		originalSize.Y.Scale * scaleFactor,
		originalSize.Y.Offset * scaleFactor
	)

	local up = TweenService:Create(obj, TweenInfo.new(duration / 2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Size = grownSize })
	local down = TweenService:Create(obj, TweenInfo.new(duration / 2, Enum.EasingStyle.Quad, Enum.EasingDirection.In), { Size = originalSize })

	local function cancel()
		up:Cancel()
		down:Cancel()
		obj.Size = originalSize
	end

	up.Completed:Connect(function(state)
		if state == Enum.PlaybackState.Completed then
			down:Play()
		end
	end)
	down.Completed:Connect(function()
		clearIfCurrent(obj, cancel)
	end)

	setActive(obj, cancel)
	up:Play()
end

--------------------------------------------------------------------------
-- FadeIn / FadeOut : apparition / disparition en douceur
--------------------------------------------------------------------------
export type FadeOptions = {
	duration: number?, -- défaut 0.3
}

local function getTransparencyProperty(obj: Instance): string?
	if obj:IsA("Frame") or obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
		return "BackgroundTransparency"
	elseif obj:IsA("TextLabel") or obj:IsA("TextButton") then
		return "TextTransparency"
	end
	return nil
end

function UIAnim.FadeIn(obj: GuiObject, options: FadeOptions?)
	options = options or {}
	local duration = options.duration or 0.3

	obj.Visible = true
	local prop = getTransparencyProperty(obj)
	local goal: { [string]: any } = {}
	if prop then
		goal[prop] = 0
	end

	local tween = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), goal)
	local function cancel()
		tween:Cancel()
	end
	tween.Completed:Connect(function()
		clearIfCurrent(obj, cancel)
	end)
	setActive(obj, cancel)
	tween:Play()
end

function UIAnim.FadeOut(obj: GuiObject, options: FadeOptions?)
	options = options or {}
	local duration = options.duration or 0.3

	local prop = getTransparencyProperty(obj)
	local goal: { [string]: any } = {}
	if prop then
		goal[prop] = 1
	end

	local tween = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Sine, Enum.EasingDirection.In), goal)
	local function cancel()
		tween:Cancel()
	end
	tween.Completed:Connect(function(state)
		if state == Enum.PlaybackState.Completed then
			obj.Visible = false
		end
		clearIfCurrent(obj, cancel)
	end)
	setActive(obj, cancel)
	tween:Play()
end

-- SlideIn : entre depuis un bord de l'écran
--------------------------------------------------------------------------
export type SlideDirection = "Top" | "Bottom" | "Left" | "Right"
export type SlideOptions = {
	direction: SlideDirection?, -- défaut "Bottom"
	distance: number?, -- pixels, défaut 60
	duration: number?, -- défaut 0.35
}

function UIAnim.SlideIn(obj: GuiObject, options: SlideOptions?)
	options = options or {}
	local direction = options.direction or "Bottom"
	local distance = options.distance or 60
	local duration = options.duration or 0.35

	local targetPosition = obj.Position
	local offsetX, offsetY = 0, 0
	if direction == "Top" then
		offsetY = -distance
	elseif direction == "Bottom" then
		offsetY = distance
	elseif direction == "Left" then
		offsetX = -distance
	elseif direction == "Right" then
		offsetX = distance
	end

	obj.Position = targetPosition + UDim2.fromOffset(offsetX, offsetY)
	obj.Visible = true

	local tween = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Position = targetPosition })
	local function cancel()
		tween:Cancel()
		obj.Position = targetPosition
	end
	tween.Completed:Connect(function()
		clearIfCurrent(obj, cancel)
	end)
	setActive(obj, cancel)
	tween:Play()
end

-- Flash : clignote (ex: alerte, surbrillance temporaire)
--------------------------------------------------------------------------
export type FlashOptions = {
	color: Color3?, -- couleur du flash, défaut rouge
	count: number?, -- nombre de clignotements, défaut 2
	duration: number?, -- durée d'un clignotement, défaut 0.15
}

function UIAnim.Flash(obj: GuiObject, options: FlashOptions?)
	options = options or {}
	local flashColor = options.color or Color3.fromRGB(255, 70, 70)
	local count = options.count or 2
	local duration = options.duration or 0.15

	if not (obj:IsA("Frame") or obj:IsA("ImageLabel") or obj:IsA("ImageButton") or obj:IsA("TextButton") or obj:IsA("TextLabel")) then
		return
	end

	local originalColor = (obj :: any).BackgroundColor3
	local cancelled = false

	local function cancel()
		cancelled = true
		;(obj :: any).BackgroundColor3 = originalColor
	end
	setActive(obj, cancel)

	task.spawn(function()
		for _ = 1, count do
			if cancelled then
				return
			end
			local toFlash = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Sine), { BackgroundColor3 = flashColor })
			toFlash:Play()
			toFlash.Completed:Wait()
			if cancelled then
				return
			end
			local toOriginal = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Sine), { BackgroundColor3 = originalColor })
			toOriginal:Play()
			toOriginal.Completed:Wait()
		end
		clearIfCurrent(obj, cancel)
	end)
end

--------------------------------------------------------------------------
-- Typewriter : révèle un texte lettre par lettre
--------------------------------------------------------------------------
export type TypewriterOptions = {
	charsPerSecond: number?, -- défaut 30
}

function UIAnim.Typewriter(label: TextLabel | TextButton, fullText: string, options: TypewriterOptions?)
	options = options or {}
	local cps = options.charsPerSecond or 30

	label.Text = ""
	local cancelled = false
	local function cancel()
		cancelled = true
		label.Text = fullText
	end
	setActive(label, cancel)

	task.spawn(function()
		local charDelay = 1 / cps
		for i = 1, #fullText do
			if cancelled then
				return
			end
			label.Text = string.sub(fullText, 1, i)
			task.wait(charDelay)
		end
		clearIfCurrent(label, cancel)
	end)
end

-- Rainbow : fait défiler les couleurs (ex: effet néon/fun)
--------------------------------------------------------------------------
export type RainbowOptions = {
	speed: number?, -- cycles par seconde, défaut 0.3
	target: ("Background" | "Text" | "Stroke")?, -- défaut "Background"
}

function UIAnim.Rainbow(obj: GuiObject, options: RainbowOptions?)
	options = options or {}
	local speed = options.speed or 0.3
	local target = options.target or "Background"

	local conn: RBXScriptConnection
	local hue = 0

	local function cancel()
		if conn then
			conn:Disconnect()
		end
	end

	conn = RunService.RenderStepped:Connect(function(dt)
		hue = (hue + dt * speed) % 1
		local color = Color3.fromHSV(hue, 0.85, 1)
		if target == "Background" and (obj :: any):FindFirstAncestorOrSelf then
			pcall(function()
				(obj :: any).BackgroundColor3 = color
			end)
		elseif target == "Text" then
			pcall(function()
				(obj :: any).TextColor3 = color
			end)
		elseif target == "Stroke" then
			local stroke = obj:FindFirstChildOfClass("UIStroke")
			if stroke then
				stroke.Color = color
			end
		end
	end)

	setActive(obj, cancel)
end

--------------------------------------------------------------------------
-- Glow : halo (UIStroke) qui pulse en continu (ex: élément actif/disponible)
--------------------------------------------------------------------------
export type GlowOptions = {
	color: Color3?, -- défaut blanc
	speed: number?, -- cycles par seconde, défaut 0.8
	maxThickness: number?, -- défaut 3
}

function UIAnim.Glow(obj: GuiObject, options: GlowOptions?)
	options = options or {}
	local color = options.color or Color3.fromRGB(255, 255, 255)
	local speed = options.speed or 0.8
	local maxThickness = options.maxThickness or 3

	local stroke = obj:FindFirstChildOfClass("UIStroke")
	local createdStroke = false
	if not stroke then
		stroke = Instance.new("UIStroke")
		stroke.Thickness = 0
		stroke.Parent = obj
		createdStroke = true
	end

	local existingStroke = stroke :: UIStroke
	local elapsed = 0
	local conn: RBXScriptConnection

	local function cancel()
		if conn then
			conn:Disconnect()
		end
		if createdStroke then
			existingStroke:Destroy()
		else
			existingStroke.Thickness = 0
		end
	end

	conn = RunService.RenderStepped:Connect(function(dt)
		elapsed += dt
		local pulse = (math.sin(elapsed * speed * math.pi * 2) + 1) / 2
		existingStroke.Color = color
		existingStroke.Thickness = pulse * maxThickness
	end)

	setActive(obj, cancel)
end

-- Ripple : cercle qui s'étend et disparaît (ex: feedback de clic)
--------------------------------------------------------------------------
export type RippleOptions = {
	color: Color3?, -- défaut blanc
	duration: number?, -- défaut 0.5
	startScale: number?, -- taille de départ (fraction du plus petit côté), défaut 0.1
}

function UIAnim.Ripple(obj: GuiObject, options: RippleOptions?)
	options = options or {}
	local color = options.color or Color3.fromRGB(255, 255, 255)
	local duration = options.duration or 0.5
	local startScale = options.startScale or 0.1

	local wasClipping = obj.ClipsDescendants
	obj.ClipsDescendants = true

	local size = math.min(obj.AbsoluteSize.X, obj.AbsoluteSize.Y)
	local circle = Instance.new("Frame")
	circle.AnchorPoint = Vector2.new(0.5, 0.5)
	circle.Position = UDim2.fromScale(0.5, 0.5)
	circle.Size = UDim2.fromOffset(size * startScale, size * startScale)
	circle.BackgroundColor3 = color
	circle.BackgroundTransparency = 0.4
	circle.BorderSizePixel = 0
	circle.ZIndex = obj.ZIndex + 1
	circle.Parent = obj

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(1, 0)
	corner.Parent = circle

	local tween = TweenService:Create(circle, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
		Size = UDim2.fromOffset(size * 1.6, size * 1.6),
		BackgroundTransparency = 1,
	})

	tween.Completed:Connect(function()
		circle:Destroy()
		obj.ClipsDescendants = wasClipping
	end)
	tween:Play()
end

-- CountUp : anime un nombre affiché dans un TextLabel (score, monnaie, XP)
--------------------------------------------------------------------------
export type CountUpOptions = {
	duration: number?, -- défaut 1
	decimals: number?, -- défaut 0
	prefix: string?, -- défaut ""
	suffix: string?, -- défaut ""
}

function UIAnim.CountUp(label: TextLabel | TextButton, from: number, to: number, options: CountUpOptions?)
	options = options or {}
	local duration = options.duration or 1
	local decimals = options.decimals or 0
	local prefix = options.prefix or ""
	local suffix = options.suffix or ""

	local function format(value: number): string
		return prefix .. string.format("%." .. decimals .. "f", value) .. suffix
	end

	local elapsed = 0
	local conn: RBXScriptConnection

	local function cancel()
		if conn then
			conn:Disconnect()
		end
		label.Text = format(to)
	end

	conn = RunService.RenderStepped:Connect(function(dt)
		elapsed += dt
		local t = math.clamp(elapsed / duration, 0, 1)
		local eased = 1 - (1 - t) * (1 - t)
		label.Text = format(from + (to - from) * eased)
		if t >= 1 then
			if conn then
				conn:Disconnect()
			end
			clearIfCurrent(label, cancel)
		end
	end)

	setActive(label, cancel)
end

-- Scramble : effet de décodage, caractères aléatoires qui se fixent
--------------------------------------------------------------------------
export type ScrambleOptions = {
	duration: number?, -- défaut 0.6
	charset: string?, -- défaut lettres majuscules + chiffres
}

local DEFAULT_SCRAMBLE_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"

function UIAnim.Scramble(label: TextLabel | TextButton, finalText: string, options: ScrambleOptions?)
	options = options or {}
	local duration = options.duration or 0.6
	local charset = options.charset or DEFAULT_SCRAMBLE_CHARS

	local length = #finalText
	local elapsed = 0
	local conn: RBXScriptConnection

	local function randomChar(): string
		local i = math.random(1, #charset)
		return string.sub(charset, i, i)
	end

	local function cancel()
		if conn then
			conn:Disconnect()
		end
		label.Text = finalText
	end

	conn = RunService.RenderStepped:Connect(function(dt)
		elapsed += dt
		local revealCount = math.floor((elapsed / duration) * length)
		if revealCount >= length then
			if conn then
				conn:Disconnect()
			end
			label.Text = finalText
			clearIfCurrent(label, cancel)
			return
		end
		local builder = {}
		for i = 1, length do
			if i <= revealCount then
				table.insert(builder, string.sub(finalText, i, i))
			else
				table.insert(builder, randomChar())
			end
		end
		label.Text = table.concat(builder)
	end)

	setActive(label, cancel)
end

-- WaveText : fait entrer un texte lettre par lettre avec un rebond en vague
--------------------------------------------------------------------------
export type WaveTextOptions = {
	staggerDelay: number?, -- délai entre chaque lettre, défaut 0.04
	bounceHeight: number?, -- pixels, défaut 10
	duration: number?, -- durée du rebond par lettre, défaut 0.3
	color: Color3?, -- défaut blanc
	textSize: number?, -- défaut 24
}

function UIAnim.WaveText(container: Frame, text: string, options: WaveTextOptions?)
	options = options or {}
	local staggerDelay = options.staggerDelay or 0.04
	local bounceHeight = options.bounceHeight or 10
	local duration = options.duration or 0.3
	local color = options.color or Color3.new(1, 1, 1)
	local textSize = options.textSize or 24

	container:ClearAllChildren()

	local layout = Instance.new("UIListLayout")
	layout.FillDirection = Enum.FillDirection.Horizontal
	layout.SortOrder = Enum.SortOrder.LayoutOrder
	layout.Parent = container

	local cancelled = false
	local letters: { TextLabel } = {}

	for i = 1, #text do
		local char = string.sub(text, i, i)
		local charLabel = Instance.new("TextLabel")
		charLabel.Name = "Char" .. i
		charLabel.LayoutOrder = i
		charLabel.Size = UDim2.fromOffset(0, container.AbsoluteSize.Y)
		charLabel.AutomaticSize = Enum.AutomaticSize.X
		charLabel.BackgroundTransparency = 1
		charLabel.Text = char
		charLabel.TextColor3 = color
		charLabel.TextSize = textSize
		charLabel.Font = Enum.Font.GothamBold
		charLabel.Parent = container
		table.insert(letters, charLabel)
	end

	local function cancel()
		cancelled = true
	end
	setActive(container, cancel)

	task.spawn(function()
		for _, charLabel in ipairs(letters) do
			if cancelled then
				return
			end
			local startPos = charLabel.Position
			charLabel.Position = startPos - UDim2.fromOffset(0, bounceHeight)
			TweenService:Create(charLabel, TweenInfo.new(duration, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Position = startPos }):Play()
			task.wait(staggerDelay)
		end
		clearIfCurrent(container, cancel)
	end)
end

-- Glitch : léger effet de distorsion numérique (ex: erreur, ambiance sci-fi)
--------------------------------------------------------------------------
export type GlitchOptions = {
	duration: number?, -- défaut 0.4
	intensity: number?, -- pixels, défaut 4
	glitchColor: Color3?, -- défaut cyan
}

function UIAnim.Glitch(label: TextLabel | TextButton, options: GlitchOptions?)
	options = options or {}
	local duration = options.duration or 0.4
	local intensity = options.intensity or 4
	local glitchColor = options.glitchColor or Color3.fromRGB(80, 230, 255)

	local originalPosition = label.Position
	local originalColor = label.TextColor3
	local elapsed = 0
	local conn: RBXScriptConnection

	local function cancel()
		if conn then
			conn:Disconnect()
		end
		label.Position = originalPosition
		label.TextColor3 = originalColor
	end

	conn = RunService.RenderStepped:Connect(function(dt)
		elapsed += dt
		if elapsed >= duration then
			cancel()
			clearIfCurrent(label, cancel)
			return
		end
		if math.random() < 0.5 then
			local offsetX = (math.random() * 2 - 1) * intensity
			label.Position = originalPosition + UDim2.fromOffset(offsetX, 0)
			label.TextColor3 = glitchColor
		else
			label.Position = originalPosition
			label.TextColor3 = originalColor
		end
	end)

	setActive(label, cancel)
end

-- MenuIn : ouverture d'un menu (scale + fade, avec fond optionnel)
--------------------------------------------------------------------------
export type MenuInOptions = {
	duration: number?, -- défaut 0.35
	fromScale: number?, -- défaut 0.85
	backdrop: GuiObject?, -- Frame de fond à estomper en même temps
	backdropTransparency: number?, -- transparence finale du fond, défaut 0.5
}

function UIAnim.MenuIn(panel: GuiObject, options: MenuInOptions?)
	options = options or {}
	local duration = options.duration or 0.35
	local fromScale = options.fromScale or 0.85
	local backdrop = options.backdrop
	local backdropTransparency = options.backdropTransparency or 0.5

	local targetSize = panel.Size
	local shrunkSize = UDim2.new(
		targetSize.X.Scale * fromScale, targetSize.X.Offset * fromScale,
		targetSize.Y.Scale * fromScale, targetSize.Y.Offset * fromScale
	)

	panel.Size = shrunkSize
	panel.Visible = true

	local panelProp = getTransparencyProperty(panel)
	local goal: { [string]: any } = { Size = targetSize }
	if panelProp then
		(panel :: any)[panelProp] = 1
		goal[panelProp] = 0
	end

	local tween = TweenService:Create(panel, TweenInfo.new(duration, Enum.EasingStyle.Back, Enum.EasingDirection.Out), goal)

	local backdropTween: Tween? = nil
	if backdrop then
		backdrop.Visible = true
		backdrop.BackgroundTransparency = 1
		backdropTween = TweenService:Create(backdrop, TweenInfo.new(duration, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), { BackgroundTransparency = backdropTransparency })
	end

	local function cancel()
		tween:Cancel()
		if backdropTween then
			backdropTween:Cancel()
		end
	end

	tween.Completed:Connect(function()
		clearIfCurrent(panel, cancel)
	end)

	setActive(panel, cancel)
	tween:Play()
	if backdropTween then
		backdropTween:Play()
	end
end

--------------------------------------------------------------------------
-- MenuOut : fermeture d'un menu (scale + fade inverse, caché à la fin)
--------------------------------------------------------------------------
export type MenuOutOptions = {
	duration: number?, -- défaut 0.25
	toScale: number?, -- défaut 0.85
	backdrop: GuiObject?,
	onComplete: (() -> ())?,
}

function UIAnim.MenuOut(panel: GuiObject, options: MenuOutOptions?)
	options = options or {}
	local duration = options.duration or 0.25
	local toScale = options.toScale or 0.85
	local backdrop = options.backdrop
	local onComplete = options.onComplete

	local originalSize = panel.Size
	local shrunkSize = UDim2.new(
		originalSize.X.Scale * toScale, originalSize.X.Offset * toScale,
		originalSize.Y.Scale * toScale, originalSize.Y.Offset * toScale
	)

	local panelProp = getTransparencyProperty(panel)
	local goal: { [string]: any } = { Size = shrunkSize }
	if panelProp then
		goal[panelProp] = 1
	end

	local tween = TweenService:Create(panel, TweenInfo.new(duration, Enum.EasingStyle.Back, Enum.EasingDirection.In), goal)

	local backdropTween: Tween? = nil
	if backdrop then
		backdropTween = TweenService:Create(backdrop, TweenInfo.new(duration, Enum.EasingStyle.Sine, Enum.EasingDirection.In), { BackgroundTransparency = 1 })
	end

	local function cancel()
		tween:Cancel()
		if backdropTween then
			backdropTween:Cancel()
		end
	end

	tween.Completed:Connect(function(state)
		if state == Enum.PlaybackState.Completed then
			panel.Visible = false
			panel.Size = originalSize
			if panelProp then
				(panel :: any)[panelProp] = 0
			end
			if backdrop then
				backdrop.Visible = false
			end
			if onComplete then
				onComplete()
			end
		end
		clearIfCurrent(panel, cancel)
	end)

	setActive(panel, cancel)
	tween:Play()
	if backdropTween then
		backdropTween:Play()
	end
end

--------------------------------------------------------------------------
-- MenuLoop : respiration douce en boucle tant qu'un menu reste ouvert
--------------------------------------------------------------------------
export type MenuLoopOptions = {
	scale: number?, -- amplitude de l'échelle, défaut 1.015
	speed: number?, -- cycles par seconde, défaut 0.25
}

function UIAnim.MenuLoop(obj: GuiObject, options: MenuLoopOptions?)
	options = options or {}
	local scaleAmplitude = options.scale or 1.015
	local speed = options.speed or 0.25

	local originalSize = obj.Size
	local elapsed = 0
	local conn: RBXScriptConnection

	local function cancel()
		if conn then
			conn:Disconnect()
		end
		obj.Size = originalSize
	end

	conn = RunService.RenderStepped:Connect(function(dt)
		elapsed += dt
		local pulse = (math.sin(elapsed * speed * math.pi * 2) + 1) / 2
		local factor = 1 + (scaleAmplitude - 1) * pulse
		obj.Size = UDim2.new(
			originalSize.X.Scale * factor, originalSize.X.Offset * factor,
			originalSize.Y.Scale * factor, originalSize.Y.Offset * factor
		)
	end)

	setActive(obj, cancel)
end
